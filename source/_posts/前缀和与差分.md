# 前缀和与差分 📊

## 📚 前缀和基础

### 核心概念

- **定义**：前缀和是指数组从第一个元素到当前位置的累加和
- **用途**：快速求区间和，将O(n)的区间查询优化到O(1)
- **时间复杂度**：
  - 预处理（建前缀和）：O(n)
  - 区间查询：O(1)
  - 单点修改：O(n)（需重新计算）
- **空间复杂度**：O(n)

### 一维前缀和

#### 定义与公式

```cpp
// 前缀和数组定义
// sum[i] = a[1] + a[2] + ... + a[i]

// 递推公式
sum[i] = sum[i-1] + a[i]

// 区间和公式
// sum[l,r] = a[l] + a[l+1] + ... + a[r]
//          = sum[r] - sum[l-1]
```

#### 基础实现

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 50;
int a[N], sum[N];
int n, m;

// 构建前缀和数组
void build_prefix_sum() {
    sum[0] = 0;  // 哨兵位置
    for (int i = 1; i <= n; i++) {
        sum[i] = sum[i-1] + a[i];
    }
}

// 区间查询：返回[l, r]的和
int range_query(int l, int r) {
    return sum[r] - sum[l-1];
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    
    build_prefix_sum();
    
    while (m--) {
        int l, r;
        cin >> l >> r;
        cout << range_query(l, r) << "\n";
    }
    
    return 0;
}
```

#### 应用示例

```cpp
// 示例：求数组所有子数组的和的最大值
int max_subarray_sum() {
    int max_sum = INT_MIN;
    for (int i = 1; i <= n; i++) {
        for (int j = i; j <= n; j++) {
            int current_sum = sum[j] - sum[i-1];
            max_sum = max(max_sum, current_sum);
        }
    }
    return max_sum;
}
```

### 二维前缀和

#### 定义与公式

```cpp
// 二维前缀和定义
// sum[i][j] = a[1][1] + a[1][2] + ... + a[1][j]
//           + a[2][1] + a[2][2] + ... + a[2][j]
//           + ...
//           + a[i][1] + a[i][2] + ... + a[i][j]

// 递推公式
sum[i][j] = sum[i-1][j] + sum[i][j-1] - sum[i-1][j-1] + a[i][j]

// 矩形和公式
// sum[x1,y1,x2,y2] = sum[x2][y2] - sum[x1-1][y2] - sum[x2][y1-1] + sum[x1-1][y1-1]
```

#### 基础实现

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 1e3 + 50;
int a[N][N], sum[N][N];
int n, m, q;

// 构建二维前缀和
void build_2d_prefix() {
    memset(sum, 0, sizeof(sum));
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            sum[i][j] = sum[i-1][j] + sum[i][j-1] - sum[i-1][j-1] + a[i][j];
        }
    }
}

// 矩形查询：返回(x1,y1)到(x2,y2)的和
int rect_query(int x1, int y1, int x2, int y2) {
    return sum[x2][y2] - sum[x1-1][y2] - sum[x2][y1-1] + sum[x1-1][y1-1];
}

int main() {
    cin >> n >> m >> q;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> a[i][j];
        }
    }
    
    build_2d_prefix();
    
    while (q--) {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        cout << rect_query(x1, y1, x2, y2) << "\n";
    }
    
    return 0;
}
```

#### 可视化解释

```
矩形区域 (x1,y1) 到 (x2,y2) 的和：

    y1         y2
    |          |
    v          v
x1 ┌──────────┐
   │  目标区域 │
x2 └──────────┘

sum[x2][y2] = 整个大矩形
- sum[x1-1][y2] = 去掉上方
- sum[x2][y1-1] = 去掉左方
+ sum[x1-1][y1-1] = 补回重复减去的左上角
```

## 🔍 差分基础

### 核心概念

- **定义**：差分是前缀和的逆运算，记录相邻元素的差值
- **用途**：快速进行区间修改，将O(n)的区间更新优化到O(1)
- **时间复杂度**：
  - 预处理（建差分数组）：O(n)
  - 区间更新：O(1)
  - 还原原数组：O(n)
- **空间复杂度**：O(n)

### 一维差分

#### 定义与公式

```cpp
// 差分数组定义
// diff[i] = a[i] - a[i-1]  (i >= 2)
// diff[1] = a[1]

// 性质
// a[i] = diff[1] + diff[2] + ... + diff[i] = sum[i]（差分数组的前缀和）

// 区间更新
// 将区间[l, r]的每个元素都加val
// diff[l] += val
// diff[r+1] -= val
```

#### 基础实现

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 50;
int a[N], diff[N];
int n, m;

// 构建差分数组
void build_diff() {
    diff[1] = a[1];
    for (int i = 2; i <= n; i++) {
        diff[i] = a[i] - a[i-1];
    }
}

// 区间更新：[l, r]加val
void range_update(int l, int r, int val) {
    diff[l] += val;
    if (r + 1 <= n) {
        diff[r+1] -= val;
    }
}

// 还原数组（计算差分数组的前缀和）
void restore_array() {
    a[1] = diff[1];
    for (int i = 2; i <= n; i++) {
        a[i] = a[i-1] + diff[i];
    }
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    
    build_diff();
    
    while (m--) {
        int l, r, d;
        cin >> l >> r >> d;
        range_update(l, r, d);
    }
    
    restore_array();
    
    for (int i = 1; i <= n; i++) {
        cout << a[i] << " ";
    }
    
    return 0;
}
```

#### 区间更新的直观理解

```
原数组：    a1  a2  a3  a4  a5
差分数组：  d1  d2  d3  d4  d5  d6

区间[2,4]加3：
d2 += 3    ← 从位置2开始影响后续所有元素
d5 -= 3    ← 从位置5开始取消影响

还原后：    a1  a2+3  a3+3  a4+3  a5
```

### 二维差分

#### 定义与公式

```cpp
// 二维差分定义
// diff[i][j] 表示影响从(i,j)开始往右下扩展的区域

// 区间更新
// 将矩形区域(x1,y1)到(x2,y2)加val
// diff[x1][y1] += val
// diff[x2+1][y1] -= val
// diff[x1][y2+1] -= val
// diff[x2+1][y2+1] += val
```

#### 基础实现

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 1e3 + 50;
int a[N][N], diff[N][N];
int n, m, q;

// 区间更新：矩形区域(x1,y1)到(x2,y2)加val
void range_update_2d(int x1, int y1, int x2, int y2, int val) {
    diff[x1][y1] += val;
    diff[x2+1][y1] -= val;
    diff[x1][y2+1] -= val;
    diff[x2+1][y2+1] += val;
}

// 还原数组（计算二维前缀和）
void restore_2d() {
    // 先按行计算前缀和
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            diff[i][j] += diff[i][j-1];
        }
    }
    // 再按列计算前缀和
    for (int j = 1; j <= m; j++) {
        for (int i = 1; i <= n; i++) {
            diff[i][j] += diff[i-1][j];
        }
    }
    // 加到原数组上
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            a[i][j] += diff[i][j];
        }
    }
}

int main() {
    cin >> n >> m >> q;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> a[i][j];
        }
    }
    
    while (q--) {
        int x1, y1, x2, y2, d;
        cin >> x1 >> y1 >> x2 >> y2 >> d;
        range_update_2d(x1, y1, x2, y2, d);
    }
    
    restore_2d();
    
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cout << a[i][j] << " ";
        }
        cout << "\n";
    }
    
    return 0;
}
```

#### 二维差分可视化

```
矩形区域(x1,y1)到(x2,y2)加val的四个标记：

    y1          y2
    |           |
    v           v
x1  +val       -val
    ↓           ↓
x2  -val       +val

+val: 从(x1,y1)开始往右下影响
-val: 从(x2+1,y1)开始往右下取消影响
-val: 从(x1,y2+1)开始往右下取消影响
+val: 从(x2+1,y2+1)开始往右下恢复影响（因为被取消了两次）
```

## 🚀 进阶应用

### 1. 前缀和优化算法

#### 子数组问题

```cpp
// 问题：求和为k的连续子数组个数
int subarray_sum_equals_k(int k) {
    unordered_map<int, int> cnt;
    cnt[0] = 1;  // 前缀和为0出现1次
    
    int prefix_sum = 0, ans = 0;
    for (int i = 1; i <= n; i++) {
        prefix_sum += a[i];
        ans += cnt[prefix_sum - k];  // 统计前缀和为prefix_sum-k的次数
        cnt[prefix_sum]++;
    }
    return ans;
}

// 问题：最长子数组，其和不超过k
int max_length_subarray_sum_le_k(int k) {
    int prefix_sum = 0, max_len = 0;
    deque<int> dq;  // 单调递增队列，存储前缀和的下标
    
    dq.push_back(0);
    for (int i = 1; i <= n; i++) {
        prefix_sum += a[i];
        
        // 移除不满足条件的元素
        while (!dq.empty() && prefix_sum - sum[dq.front()] > k) {
            dq.pop_front();
        }
        
        // 更新最大长度
        if (!dq.empty()) {
            max_len = max(max_len, i - dq.front());
        }
        
        // 维护单调性
        while (!dq.empty() && prefix_sum <= sum[dq.back()]) {
            dq.pop_back();
        }
        dq.push_back(i);
    }
    return max_len;
}
```

#### 前缀和与模运算

```cpp
// 问题：求最长的子数组，其元素和能被k整除
int max_length_divisible_by_k(int k) {
    unordered_map<int, int> first_pos;
    first_pos[0] = 0;  // 前缀和模k为0的最小位置是0
    
    int prefix_sum = 0, max_len = 0;
    for (int i = 1; i <= n; i++) {
        prefix_sum += a[i];
        int mod = ((prefix_sum % k) + k) % k;  // 处理负数
        
        if (first_pos.find(mod) == first_pos.end()) {
            first_pos[mod] = i;  // 第一次出现，记录位置
        } else {
            max_len = max(max_len, i - first_pos[mod]);
        }
    }
    return max_len;
}
```

### 2. 差分进阶技巧

#### 差分与前缀和结合

```cpp
// 问题：支持两种操作
// 1. 区间[l, r]加val
// 2. 查询某个位置的值
// 使用差分数组 + 树状数组（或线段树）

// 差分数组的树状数组实现
ll tree[N];

void update_diff(int pos, ll val) {
    for (; pos <= n; pos += pos & (-pos)) {
        tree[pos] += val;
    }
}

// 区间更新
void range_update_bit(int l, int r, ll val) {
    update_diff(l, val);
    update_diff(r + 1, -val);
}

// 单点查询
ll point_query_bit(int pos) {
    ll sum = 0;
    for (; pos > 0; pos -= pos & (-pos)) {
        sum += tree[pos];
    }
    return sum;
}
```

```cpp
// 问题：多次区间更新后，求最终数组
```

### 3. 前缀和与哈希表

```cpp
// 问题：给定两个数组，求最长公共子数组的长度
int longest_common_subarray(vector<int>& arr1, vector<int>& arr2) {
    int n = arr1.size(), m = arr2.size();
    unordered_map<int, vector<int>> pos_map;
    
    // 记录arr2中每个值出现的位置
    for (int j = 0; j < m; j++) {
        pos_map[arr2[j]].push_back(j);
    }
    
    // 计算arr1的前缀和
    vector<long long> sum1(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        sum1[i] = sum1[i-1] + arr1[i-1];
    }
    
    // 计算arr2的前缀和
    vector<long long> sum2(m + 1, 0);
    for (int j = 1; j <= m; j++) {
        sum2[j] = sum2[j-1] + arr2[j-1];
    }
    
    // 检查所有可能的长度
    int max_len = 0;
    for (int len = min(n, m); len >= 1; len--) {
        unordered_set<long long> sums;
        
        // 存储arr1中所有长度为len的子数组和
        for (int i = len; i <= n; i++) {
            long long s = sum1[i] - sum1[i-len];
            sums.insert(s);
        }
        
        // 检查arr2中是否有相同的子数组和
        for (int j = len; j <= m; j++) {
            long long s = sum2[j] - sum2[j-len];
            if (sums.find(s) != sums.end()) {
                return len;  // 找到最长的
            }
        }
    }
    
    return 0;
}
```

### 4. 二维前缀和的高级应用

#### 最大子矩阵和

```cpp
// 问题：求最大的子矩阵和（二维的最大子数组和）
int max_submatrix_sum() {
    int max_sum = INT_MIN;
    
    // 枚举上边界
    for (int i = 1; i <= n; i++) {
        int temp[N] = {0};
        
        // 枚举下边界
        for (int j = i; j <= n; j++) {
            // 将第i行到第j行压缩成一维数组
            for (int k = 1; k <= m; k++) {
                temp[k] += a[j][k];
            }
            
            // 在一维数组上求最大子数组和
            int current_sum = 0, max_temp = INT_MIN;
            for (int k = 1; k <= m; k++) {
                current_sum = max(temp[k], current_sum + temp[k]);
                max_temp = max(max_temp, current_sum);
            }
            
            max_sum = max(max_sum, max_temp);
        }
    }
    
    return max_sum;
}
```

#### 统计矩阵中值为1的连通区域

```cpp
// 结合前缀和统计不同区域的特征
void count_regions() {
    // 先用DFS/BFS找出所有连通区域
    // 然后用前缀和快速统计每个区域的特征
    
    vector<vector<int>> prefix_sum(n+1, vector<int>(m+1, 0));
    
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            prefix_sum[i][j] = prefix_sum[i-1][j] + prefix_sum[i][j-1] 
                             - prefix_sum[i-1][j-1] + a[i][j];
        }
    }
    
    // 快速查询任意矩形区域的和
    int query(int x1, int y1, int x2, int y2) {
        return prefix_sum[x2][y2] - prefix_sum[x1-1][y2] 
             - prefix_sum[x2][y1-1] + prefix_sum[x1-1][y1-1];
    }
}
```

### 操作复杂度对比

| 操作 | 暴力法 | 前缀和 | 差分 |

| 场景 | 推荐方法 |

## 🎊 总结

### 前缀和的核心价值

1. **快速查询**：O(1)时间查询任意区间和
2. **简单高效**：代码简洁，易于实现
3. **扩展性强**：可扩展到二维、三维

### 差分的核心价值

1. **快速更新**：O(1)时间更新任意区间
2. **批量操作**：多次更新后统一还原
3. **空间优化**：只需O(n)额外空间

### 选择建议

- **查询多，修改少** → 前缀和
- **修改多，查询少** → 差分
- **查询和修改都多** → 树状数组/线段树
- **二维问题** → 二维前缀和/差分

**记住**：前缀和和差分是互逆的运算，理解这一点有助于灵活运用它们解决问题！
